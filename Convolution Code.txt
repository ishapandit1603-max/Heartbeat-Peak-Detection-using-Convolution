fs = 500;
t = 0:1/fs:5;
ecg = 1.5*sin(2*pi*1.2*t) + 0.5*sin(2*pi*0.8*t);

load('ecg.mat');

windowSize = 5;
kernel = ones(1, windowSize) / windowSize;

smoothed_ecg = conv(ecg, kernel, 'same');

[pks, locs] = findpeaks(smoothed_ecg, 'MinPeakHeight', 0.5);

figure;
plot(smoothed_ecg, 'g');
plot(locs, pks, 'ro');
title('Heartbeat Peak Detection');
xlabel('Samples');
ylabel('Amplitude');
legend('Smoothed ECG', 'Detected Peaks');
